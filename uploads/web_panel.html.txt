  <script>
    // Paste your actual API Gateway Invoke URL below inside the quotes
    const API_URL = "https://cb51njm077.execute-api.ap-south-1.amazonaws.com/prod-stage/backup";

    async function triggerBackup() {
      const statusDiv = document.getElementById('status');
      statusDiv.innerHTML = 'Triggering cloud pipeline...';
      statusDiv.style.color = '#d97706';
      
      try {
        const response = await fetch(API_URL, { method: 'POST' });
        const data = await response.json();
        
        if(response.ok) {
          statusDiv.innerHTML = '✅ SUCCESS: ' + data.message + '<br>Stored File: ' + data.file;
          statusDiv.style.color = 'green';
        } else {
          statusDiv.innerHTML = '❌ ERROR: ' + data.error;
          statusDiv.style.color = 'red';
        }
      } catch (err) {
        statusDiv.innerHTML = '❌ NETWORK ERROR: Verify your API Gateway URL and check if CORS is enabled.';
        statusDiv.style.color = 'red';
      }
    }
  </script>
